<script>

function optimize() {

    // Get user inputs

    const soil =
        document.getElementById("soil").value;

    const moisture =
        Number(
            document.getElementById("moisture").value
        );

    const water =
        Number(
            document.getElementById("water").value
        );

    const rain =
        Number(
            document.getElementById("rain").value
        );

    const temp =
        Number(
            document.getElementById("temp").value
        );

    const area =
        Number(
            document.getElementById("area").value
        );


    // Crop candidates

    let candidates = [

        {
            name: "Paddy",
            score: 55,
            baseWater: 220
        },

        {
            name: "Maize",
            score: 60,
            baseWater: 150
        },

        {
            name: "Groundnut",
            score: 58,
            baseWater: 120
        },

        {
            name: "Millets",
            score: 62,
            baseWater: 80
        }

    ];


    // AI-style scoring logic

    candidates.forEach(crop => {


        // Soil suitability

        if (soil === "loamy") {

            crop.score += 10;

        }


        if (
            soil === "clay" &&
            crop.name === "Paddy"
        ) {

            crop.score += 15;

        }


        if (
            soil === "sandy" &&
            (
                crop.name === "Groundnut" ||
                crop.name === "Millets"
            )
        ) {

            crop.score += 15;

        }


        // Rainfall

        if (
            rain >= 70 &&
            crop.name === "Millets"
        ) {

            crop.score += 4;

        }


        // Water availability

        if (water < 150) {

            if (
                crop.name === "Millets" ||
                crop.name === "Groundnut"
            ) {

                crop.score += 15;

            }

        }


        // Temperature

        if (
            temp >= 25 &&
            temp <= 32
        ) {

            crop.score += 5;

        }


        // Soil moisture

        if (
            moisture > 80 &&
            crop.name === "Paddy"
        ) {

            crop.score -= 5;

        }

    });


    // Select highest score

    candidates.sort(
        (a, b) => b.score - a.score
    );


    const best =
        candidates[0];


    const score =
        Math.min(100, best.score);


    // Reasons

    let reasons = [];

    reasons.push(
        "Soil type was considered for crop suitability."
    );

    reasons.push(
        "Available water was compared with crop water needs."
    );

    reasons.push(
        "Rainfall and temperature were included."
    );

    reasons.push(
        "Current soil moisture was considered before irrigation planning."
    );


    // Resource calculations

    const dailyWater =
        Math.max(
            20,
            Math.round(
                best.baseWater *
                area *
                (rain >= 60 ? 0.75 : 1)
            )
        );


    const fertilizer =
        Math.max(
            1,
            Math.round(area * 1.8)
        );


    const efficiency =
        Math.min(
            95,
            Math.round(
                65 +
                (water < 250 ? 15 : 5) +
                (rain >= 60 ? 8 : 0)
            )
        );


    // Show result

    document.getElementById(
        "result"
    ).style.display = "block";


    document.getElementById(
        "crop"
    ).textContent =
        best.name;


    document.getElementById(
        "score"
    ).textContent =
        score + "/100";


    document.getElementById(
        "fill"
    ).style.width =
        score + "%";


    document.getElementById(
        "reasons"
    ).innerHTML =
        reasons
        .map(
            reason =>
                "<li>" +
                reason +
                "</li>"
        )
        .join("");


    document.getElementById(
        "waterRec"
    ).textContent =
        dailyWater + " L";


    document.getElementById(
        "fertRec"
    ).textContent =
        fertilizer + " kg";


    document.getElementById(
        "effRec"
    ).textContent =
        efficiency + "%";


    // AI recommendation

    let action =
        "Choose " +
        best.name +
        " based on the current farm conditions. ";


    if (moisture > 75) {

        action +=
            "Reduce immediate irrigation because soil moisture is already high. ";

    } else {

        action +=
            "Use irrigation according to soil moisture instead of a fixed schedule. ";

    }


    if (rain >= 60) {

        action +=
            "Expected rainfall may reduce irrigation demand, so review the water plan before the rain.";

    } else {

        action +=
            "Continue monitoring rainfall and soil moisture and update the plan when conditions change.";

    }


    document.getElementById(
        "action"
    ).textContent =
        action;

}
